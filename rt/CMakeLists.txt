cmake_minimum_required(VERSION 3.20)
project(clarity_rt CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)

add_library(clarity_core INTERFACE)
target_include_directories(clarity_core INTERFACE include)

enable_testing()
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core clarity_core)
find_package(Threads REQUIRED)
target_link_libraries(test_core Threads::Threads)
add_test(NAME core COMMAND test_core)

# ---- ONNX Runtime engine (optional: needs onnxruntime headers/libs) ----
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS /opt/homebrew/opt/onnxruntime/include/onnxruntime
          /usr/local/include/onnxruntime /usr/include/onnxruntime ${ORT_ROOT}/include)
find_library(ORT_LIB onnxruntime PATHS /opt/homebrew/opt/onnxruntime/lib /usr/local/lib ${ORT_ROOT}/lib)
if(ORT_INCLUDE AND ORT_LIB)
  add_library(clarity_engine src/engine.cpp)
  target_include_directories(clarity_engine PUBLIC include ${ORT_INCLUDE})
  target_link_libraries(clarity_engine PUBLIC ${ORT_LIB})
  add_executable(clarity_bench src/bench_main.cpp)
  target_link_libraries(clarity_bench clarity_engine Threads::Threads)
  add_executable(clarity_rt src/rt_main.cpp)
  target_link_libraries(clarity_rt clarity_engine Threads::Threads)
  if(APPLE)
    target_link_libraries(clarity_rt "-framework AudioToolbox" "-framework CoreAudio" "-framework CoreFoundation")
  endif()
else()
  message(STATUS "onnxruntime not found: building core tests only")
endif()
