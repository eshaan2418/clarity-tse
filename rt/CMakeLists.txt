cmake_minimum_required(VERSION 3.20)
project(clarity_rt CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)

add_library(clarity_core INTERFACE)
target_include_directories(clarity_core INTERFACE include)

enable_testing()
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core clarity_core)
find_package(Threads REQUIRED)
target_link_libraries(test_core Threads::Threads)
add_test(NAME core COMMAND test_core)
